SQL:
SET @boundary = 'capacityS4-prepare-r90-20260905T0610Z-prepare-';
SELECT 'support_turn_decision' AS evidence,
       CAST(SUBSTRING(SUBSTRING_INDEX(SUBSTRING(CAST(t.correlation_key AS CHAR),
         CHAR_LENGTH(@boundary) + 1), '-', 1), 6) AS UNSIGNED) AS rate,
       t.state, COALESCE(t.outcome,'NULL') AS outcome,
       COALESCE(t.failure_code,'NULL') AS failure_code, COUNT(*) AS row_count
FROM cs_db.support_turn t
WHERE LEFT(CAST(t.correlation_key AS BINARY), OCTET_LENGTH(@boundary))
      = CAST(@boundary AS BINARY)
GROUP BY rate, t.state, t.outcome, t.failure_code
ORDER BY rate, t.state, t.outcome, t.failure_code;
SELECT 'commerce_pending_action' AS evidence,
       CAST(SUBSTRING(SUBSTRING_INDEX(SUBSTRING(CAST(t.correlation_key AS CHAR),
         CHAR_LENGTH(@boundary) + 1), '-', 1), 6) AS UNSIGNED) AS rate,
       p.state, COUNT(*) AS row_count
FROM cs_db.support_turn t JOIN commerce_db.pending_action p ON p.turn_id=t.turn_id
WHERE LEFT(CAST(t.correlation_key AS BINARY), OCTET_LENGTH(@boundary))
      = CAST(@boundary AS BINARY)
GROUP BY rate, p.state ORDER BY rate;
SELECT 'agent_pending_reference' AS evidence,
       CAST(SUBSTRING(SUBSTRING_INDEX(SUBSTRING(CAST(t.correlation_key AS CHAR),
         CHAR_LENGTH(@boundary) + 1), '-', 1), 6) AS UNSIGNED) AS rate,
       r.state, COUNT(*) AS row_count
FROM cs_db.support_turn t JOIN cs_db.pending_action_reference r ON r.source_turn_id=t.turn_id
WHERE LEFT(CAST(t.correlation_key AS BINARY), OCTET_LENGTH(@boundary))
      = CAST(@boundary AS BINARY)
GROUP BY rate, r.state ORDER BY rate;
SELECT 'action_receipt' AS evidence, COUNT(*) AS row_count
FROM commerce_db.action_receipt a
JOIN commerce_db.pending_action p USING (pending_action_id)
JOIN cs_db.support_turn t ON t.turn_id=p.turn_id
WHERE LEFT(CAST(t.correlation_key AS BINARY), OCTET_LENGTH(@boundary))
      = CAST(@boundary AS BINARY);
